# Syntax: the interface of languages and their syntax trees, the registry of languages and category names, and the
# tree-sitter languages that the build turns on.
pr_library(syntax INCLUDE include
    src/syntax.cpp
)
target_link_libraries(pr_syntax PUBLIC prax::core prax::text_edit PRIVATE prax::core_cpp prax::regex)

# Adds the tree-sitter grammar of `language` to the syntax library: a library of the grammar's sources, its source
# in the backend (syntax_treesitter_<language>.cpp), and a header, generated here, that holds the grammar's queries.
# The queries of each list are joined in order, as tree-sitter's tools join them.
#
#   pr_syntax_grammar(<language> SOURCES <file>... INCLUDE <directory>... [HIGHLIGHTS <file>...]
#       [INJECTIONS <file>...])
function(pr_syntax_grammar language)
    cmake_parse_arguments(PARSE_ARGV 1 GRAMMAR "" "" "SOURCES;INCLUDE;HIGHLIGHTS;INJECTIONS")
    set(target pr_tree_sitter_${language})
    add_library(${target} STATIC ${GRAMMAR_SOURCES})
    target_include_directories(${target} PRIVATE ${GRAMMAR_INCLUDE})
    pr_third_party(${target})
    set_property(GLOBAL APPEND PROPERTY PR_LIBRARIES ${target})
    target_link_libraries(pr_syntax PRIVATE ${target})
    target_include_directories(pr_syntax SYSTEM PRIVATE ${GRAMMAR_INCLUDE})
    target_sources(pr_syntax PRIVATE backends/treesitter/src/syntax_treesitter_${language}.cpp)

    foreach (kind IN ITEMS HIGHLIGHTS INJECTIONS)
        set(${kind} "")
        foreach (file IN LISTS GRAMMAR_${kind})
            file(READ ${file} text)
            string(APPEND ${kind} "${text}\n")
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
        endforeach()
    endforeach()
    set(header ${PR_SYNTAX_GENERATED}/syntax/backends/treesitter/src/syntax_queries_${language}.h)
    # Written as it is, so that nothing replaces the @ of a query's captures.
    file(WRITE ${header}.new
        "// Generated by lib/syntax/CMakeLists.txt from the queries of tree-sitter's ${language}. Do not edit.\n\n"
        "#pragma once\n\n"
        "namespace pr_impl {\n\n"
        "constexpr char kTreeSitterHighlights[] = R\"query(${HIGHLIGHTS})query\";\n"
        "constexpr char kTreeSitterInjections[] = R\"query(${INJECTIONS})query\";\n\n"
        "} // namespace pr_impl\n")
    file(COPY_FILE ${header}.new ${header} ONLY_IF_DIFFERENT)
endfunction()

if (PR_SYNTAX_TREESITTER)
    pr_dependency_available(tree_sitter)
    add_library(pr_tree_sitter STATIC ${tree_sitter_SOURCE_DIR}/lib/src/lib.c)
    add_library(prax::tree_sitter ALIAS pr_tree_sitter)
    set_target_properties(pr_tree_sitter PROPERTIES EXPORT_NAME tree_sitter)
    target_include_directories(pr_tree_sitter SYSTEM PUBLIC $<BUILD_INTERFACE:${tree_sitter_SOURCE_DIR}/lib/include>
        PRIVATE ${tree_sitter_SOURCE_DIR}/lib/src)
    # The POSIX functions that it calls, such as fdopen, as its own build declares them.
    target_compile_definitions(pr_tree_sitter PRIVATE _POSIX_C_SOURCE=200112L _DEFAULT_SOURCE)
    pr_third_party(pr_tree_sitter)
    set_property(GLOBAL APPEND PROPERTY PR_LIBRARIES pr_tree_sitter)

    target_sources(pr_syntax PRIVATE
        backends/treesitter/src/syntax_treesitter.cpp
    )
    pr_headers(pr_syntax PUBLIC backends/treesitter/include)
    target_compile_definitions(pr_syntax PRIVATE PR_SYNTAX_HAS_TREESITTER)
    target_link_libraries(pr_syntax PRIVATE pr_tree_sitter)
    set(PR_SYNTAX_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_include_directories(pr_syntax PRIVATE ${PR_SYNTAX_GENERATED})

    foreach (grammar IN ITEMS c cpp css html javascript json markdown python xml)
        pr_dependency_available(tree_sitter_${grammar})
    endforeach()
    pr_syntax_grammar(c
        SOURCES ${tree_sitter_c_SOURCE_DIR}/src/parser.c
        INCLUDE ${tree_sitter_c_SOURCE_DIR}/src ${tree_sitter_c_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_c_SOURCE_DIR}/queries/highlights.scm)
    pr_syntax_grammar(cpp
        SOURCES ${tree_sitter_cpp_SOURCE_DIR}/src/parser.c ${tree_sitter_cpp_SOURCE_DIR}/src/scanner.c
        INCLUDE ${tree_sitter_cpp_SOURCE_DIR}/src ${tree_sitter_cpp_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_c_SOURCE_DIR}/queries/highlights.scm ${tree_sitter_cpp_SOURCE_DIR}/queries/highlights.scm
        INJECTIONS ${tree_sitter_cpp_SOURCE_DIR}/queries/injections.scm)
    pr_syntax_grammar(css
        SOURCES ${tree_sitter_css_SOURCE_DIR}/src/parser.c ${tree_sitter_css_SOURCE_DIR}/src/scanner.c
        INCLUDE ${tree_sitter_css_SOURCE_DIR}/src ${tree_sitter_css_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_css_SOURCE_DIR}/queries/highlights.scm)
    pr_syntax_grammar(html
        SOURCES ${tree_sitter_html_SOURCE_DIR}/src/parser.c ${tree_sitter_html_SOURCE_DIR}/src/scanner.c
        INCLUDE ${tree_sitter_html_SOURCE_DIR}/src ${tree_sitter_html_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_html_SOURCE_DIR}/queries/highlights.scm
        INJECTIONS ${tree_sitter_html_SOURCE_DIR}/queries/injections.scm)
    pr_syntax_grammar(javascript
        SOURCES ${tree_sitter_javascript_SOURCE_DIR}/src/parser.c ${tree_sitter_javascript_SOURCE_DIR}/src/scanner.c
        INCLUDE ${tree_sitter_javascript_SOURCE_DIR}/src ${tree_sitter_javascript_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_javascript_SOURCE_DIR}/queries/highlights.scm
            ${tree_sitter_javascript_SOURCE_DIR}/queries/highlights-jsx.scm
            ${tree_sitter_javascript_SOURCE_DIR}/queries/highlights-params.scm
        INJECTIONS ${tree_sitter_javascript_SOURCE_DIR}/queries/injections.scm)
    pr_syntax_grammar(json
        SOURCES ${tree_sitter_json_SOURCE_DIR}/src/parser.c
        INCLUDE ${tree_sitter_json_SOURCE_DIR}/src ${tree_sitter_json_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_json_SOURCE_DIR}/queries/highlights.scm)
    # Markdown is two grammars: its blocks, and its inline text, which the blocks inject.
    set(markdown ${tree_sitter_markdown_SOURCE_DIR}/tree-sitter-markdown)
    set(markdownInline ${tree_sitter_markdown_SOURCE_DIR}/tree-sitter-markdown-inline)
    pr_syntax_grammar(markdown
        SOURCES ${markdown}/src/parser.c ${markdown}/src/scanner.c
        INCLUDE ${markdown}/src ${markdown}/bindings/c
        HIGHLIGHTS ${markdown}/queries/highlights.scm
        INJECTIONS ${markdown}/queries/injections.scm)
    pr_syntax_grammar(markdown_inline
        SOURCES ${markdownInline}/src/parser.c ${markdownInline}/src/scanner.c
        INCLUDE ${markdownInline}/src ${markdownInline}/bindings/c
        HIGHLIGHTS ${markdownInline}/queries/highlights.scm
        INJECTIONS ${markdownInline}/queries/injections.scm)
    pr_syntax_grammar(python
        SOURCES ${tree_sitter_python_SOURCE_DIR}/src/parser.c ${tree_sitter_python_SOURCE_DIR}/src/scanner.c
        INCLUDE ${tree_sitter_python_SOURCE_DIR}/src ${tree_sitter_python_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_python_SOURCE_DIR}/queries/highlights.scm)

        SOURCES ${tree_sitter_xml_SOURCE_DIR}/xml/src/parser.c ${tree_sitter_xml_SOURCE_DIR}/xml/src/scanner.c
        INCLUDE ${tree_sitter_xml_SOURCE_DIR}/xml/src ${tree_sitter_xml_SOURCE_DIR}/bindings/c
        HIGHLIGHTS ${tree_sitter_xml_SOURCE_DIR}/queries/xml/highlights.scm)
endif()
